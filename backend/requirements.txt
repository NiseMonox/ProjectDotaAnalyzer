uvicorn[standard]==0.30.0

# CVE-2024-53981 (multipart boundary DoS).

# AI coach LLM client (DeepSeek, OpenAI-compatible). Without it the coach
# silently falls back to rule-based reports. Capped at the major version the
# coach has made live calls with: 2.31.0 wrote the validated reports of
# 2026-10-11 (4 reports, no protocol error). A fresh install resolves to the
# newest 2.x (2.54.0 on that day), which passes the test suite on Python
# 3.11, 3.12 and 3.13. 3.28.0 also passes the suite and a round trip of the
# coach's call path against a local fake endpoint, but has not written a real
# report: widen the range after one.
openai>=1.30,<3
# Loads <project>/.env (backend/app/config.py)
python-dotenv>=1.0,<2
# Vector maths for the knowledge retrieval (dense term of the RAG ranker).
numpy>=1.26,<3

# Zstandard under the .dem.bz2 name (bzip2 uploads need only the standard
# library). No dependencies of its own. Optional at runtime: without it such
# an upload is refused with a message asking the user to decompress it first.
zstandard>=0.22,<1
